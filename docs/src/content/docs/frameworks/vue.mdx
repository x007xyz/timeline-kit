## 代码示例

```vue
<script setup lang="ts">
import { ref, shallowRef } from 'vue'
import { demoDocument, type TimelineEditor as VanillaEditor } from 'timeline-kit'
import { TimelineEditor, useTimelineSelector } from 'timeline-kit/vue'

const editorCompRef = ref<{ exec: (cmd: string) => boolean } | null>(null)
const editorInstance = shallowRef<VanillaEditor | null>(null)

const currentFrame = useTimelineSelector(editorInstance, s => s.currentFrame)
const canUndo = useTimelineSelector(editorInstance, s => s.canUndo)

function onReady(editor: VanillaEditor) {
  editorInstance.value = editor
}
</script>

<template>
  <div>当前帧: {{ currentFrame ?? 0 }} · 可撤销: {{ canUndo ? '是' : '否' }}</div>
  <div style="height: 340px">
    <TimelineEditor
      ref="editorCompRef"
      :document="demoDocument"
      theme="dark"
      @ready="onReady"
    >
      <template #clip="{ clip, isSelected }">
        <div :class="['my-clip', { selected: isSelected }]">
          {{ clip.name }}
        </div>
      </template>
    </TimelineEditor>
  </div>
</template>
```
